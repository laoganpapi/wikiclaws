\section{Introduction}
\label{sec:intro}

\TODO{Open with the $3n+1$ map and a brief narrative.}

The Collatz conjecture, posed informally by Lothar Collatz in 1937,
asserts that the iteration
\[
   \T(n) = \begin{cases} n/2 & n \text{ even}, \\ 3n+1 & n \text{ odd}, \end{cases}
\]
eventually reaches $1$ for every positive integer $n$. Despite
verification to $n \leq 2^{68}$~\cite{barina2020} and major partial results,
the conjecture remains open. The current state-of-the-art density result
is due to Tao~\cite{tao2019}, who showed that almost every orbit (in the
sense of logarithmic density) attains values that are arbitrarily small
relative to a chosen unbounded function.

\begin{definition}[Collatz map]
The \emph{Collatz map} $\T : \nat \to \nat$ is
\[
   \T(n) = \begin{cases} n/2 & n \equiv 0 \pmod 2, \\
                          3n+1 & n \equiv 1 \pmod 2. \end{cases}
\]
\end{definition}

\begin{definition}[Stopping time]
The \emph{stopping time} of $n$ is $\stoptime{n} = \min\{k \geq 1 : \Tit{k}(n) < n\}$,
and the \emph{total stopping time} is
$\totstoptime{n} = \min\{k \geq 0 : \Tit{k}(n) = 1\}$.
\end{definition}

\begin{definition}[Parity vector]
The \emph{parity vector} of $n$ at length $k$ is
$\pvk{k}{n} = (\Tit{0}(n) \bmod 2, \Tit{1}(n) \bmod 2, \ldots, \Tit{k-1}(n) \bmod 2) \in \{0,1\}^k$.
\end{definition}

\begin{conjecture}[Collatz, 1937]\label{conj:collatz}
For every $n \in \nat$, there exists $k \geq 0$ with $\Tit{k}(n) = 1$.
\end{conjecture}
