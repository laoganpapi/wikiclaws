finite \UCfam\ $\UC$ other than $\{\emptyset\}$ contains an element of the

\begin{definition}[Union-closed family]
A family $\UC \subseteq \powerset{\ground}$ is \emph{union-closed} if
$A, B \in \UC$ implies $A \cup B \in \UC$.
\end{definition}

\begin{definition}[Abundance]
For $i \in \ground$, the \emph{abundance} of $i$ in $\UC$ is
\[
   \abun_\UC(i) \;\coloneqq\; \frac{|\{A \in \UC : i \in A\}|}{|\UC|}.
\]
\end{definition}

\begin{conjecture}[Frankl, 1979]\label{conj:frankl}
For every finite union-closed family $\UC \neq \{\emptyset\}$, there exists
$i \in \ground$ with $\abun_\UC(i) \geq \target$.
\end{conjecture}
